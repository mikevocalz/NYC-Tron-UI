import { createPageWindowHost } from "@nyc-tron/core/window-chrome";
import { WindowTitleBar, useWindowChrome } from "@nyc-tron/ui/components/window/WindowTitleBar";

// In-page geometry only; this does not resize the app's system window.
export default function Example() {
  const host = React.useRef<ReturnType<typeof createPageWindowHost> | null>(null);
  if (!host.current) {
    host.current = createPageWindowHost();
    host.current.open({id:"example",displayId:"page",rect:{x:0,y:0,width:640,height:480}});
  }
  const chrome = useWindowChrome(host.current,"example");
  return <WindowTitleBar chrome={chrome} title="Example document" controls={{kind:"drawn",layout:"windows"}} />;
}
